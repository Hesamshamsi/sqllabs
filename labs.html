کار عملی ۱

طراحی پایگاه داده اطلاعات دانشجویان

آشنایی با طراحی یک پایگاه داده ساده برای ثبت، نگهداری، جستجو و ویرایش اطلاعات دانشجویان. فقط یک موجودیت (Student) داریم، پس رابطه‌ای تعریف نمی‌شود.

صفتفیلدنوع داده
شناسه دانشجو (کلید اصلی)StudentIDINT
شماره دانشجوییStudentNumberNVARCHAR(20)
نامFirstNameNVARCHAR(50)
نام خانوادگیLastNameNVARCHAR(50)
تاریخ تولدBirthDateDATE
نشانیAddressNVARCHAR(200)
شماره تماسPhoneNVARCHAR(20)

ایجاد پایگاه داده و جدول

CREATE DATABASE StudentsDB;
GO
USE StudentsDB;
GO
CREATE TABLE Students (
  StudentID INT IDENTITY(1,1) PRIMARY KEY,
  StudentNumber NVARCHAR(20) NOT NULL UNIQUE,
  FirstName NVARCHAR(50) NOT NULL,
  LastName NVARCHAR(50) NOT NULL,
  BirthDate DATE NULL,
  Address NVARCHAR(200) NULL,
  Phone NVARCHAR(20) NULL
);
GO
  • CREATE DATABASE پایگاه داده و CREATE TABLE جدول را می‌سازد.
  • PRIMARY KEY کلید اصلی است و برای هر رکورد یکتاست.
  • IDENTITY(1,1) شناسه را از ۱ شروع می‌کند و برای هر رکورد یک واحد زیاد می‌کند.
  • NOT NULL ورود مقدار را الزامی می‌کند و UNIQUE مانع تکراری شدن شماره دانشجویی می‌شود.
  • NVARCHAR برای متن فارسی مناسب است.

کار عملی ۲

مدیریت شماره تماس دانشجویان

هر دانشجو می‌تواند چند شماره تماس (همراه، منزل، اضطراری) داشته باشد؛ پس رابطه یک‌به‌چند (One-to-Many) بین Students و StudentPhones برقرار می‌شود.

Students (1) ───────< StudentPhones (N)

شماره تلفن از نوع NVARCHAR تعریف می‌شود چون مقدار محاسباتی نیست و ممکن است با صفر شروع شود.

ایجاد پایگاه داده و جداول

CREATE DATABASE StudentContactsDB;
GO
USE StudentContactsDB;
GO
CREATE TABLE Students
(
  StudentID INT IDENTITY(1,1) PRIMARY KEY,
  StudentNumber NVARCHAR(20) NOT NULL UNIQUE,
  FirstName NVARCHAR(50) NOT NULL,
  LastName NVARCHAR(50) NOT NULL
);
GO
CREATE TABLE StudentPhones
(
  PhoneID INT IDENTITY(1,1) PRIMARY KEY,
  StudentID INT NOT NULL,
  PhoneNumber NVARCHAR(20) NOT NULL,
  PhoneType NVARCHAR(30) NOT NULL,
  IsPrimary BIT NOT NULL DEFAULT 0,
  CONSTRAINT FK_StudentPhones_Students
    FOREIGN KEY (StudentID)
    REFERENCES Students(StudentID)
);
GO
  • FOREIGN KEY ارتباط دو جدول را برقرار می‌کند.
  • DEFAULT 0 باعث می‌شود شماره جدید به‌طور پیش‌فرض «اصلی» حساب نشود.
  • IsPrimary از نوع BIT است: ۱ یعنی شماره اصلی، ۰ یعنی غیر اصلی.

تمرین ۹: شمارش شماره‌های تماس هر دانشجو

SELECT
  Students.StudentNumber,
  Students.FirstName,
  Students.LastName,
  COUNT(StudentPhones.PhoneID) AS PhoneCount
FROM Students
LEFT JOIN StudentPhones
  ON Students.StudentID = StudentPhones.StudentID
GROUP BY
  Students.StudentNumber,
  Students.FirstName,
  Students.LastName;

LEFT JOIN باعث می‌شود دانشجویان بدون شماره هم نمایش داده شوند.

تمرین ۱۰: دانشجویان بدون شماره تماس

SELECT
  Students.StudentNumber,
  Students.FirstName,
  Students.LastName
FROM Students
LEFT JOIN StudentPhones
  ON Students.StudentID = StudentPhones.StudentID
WHERE StudentPhones.PhoneID IS NULL;

مدیریت دانشکده‌های دانشگاه

هر دانشگاه می‌تواند صفر، یک یا چند دانشکده داشته باشد، اما هر دانشکده فقط به یک دانشگاه تعلق دارد (رابطه یک‌به‌چند).

Universities (1) ───────< Faculties (N)
UniversitiesFaculties
UniversityID (PK, IDENTITY)
UniversityName NVARCHAR(100)
City NVARCHAR(50)
EstablishmentYear INT		 FacultyID (PK, IDENTITY)
FacultyName NVARCHAR(100)
DeanName NVARCHAR(100)
Building NVARCHAR(100)
UniversityID INT (FK)

ایجاد پایگاه داده و جداول

CREATE DATABASE UniversityFacultiesDB;
GO
USE UniversityFacultiesDB;
GO
CREATE TABLE Universities
(
  UniversityID INT IDENTITY(1,1) PRIMARY KEY,
  UniversityName NVARCHAR(100) NOT NULL,
  City NVARCHAR(50) NOT NULL,
  EstablishmentYear INT NULL
);
GO
CREATE TABLE Faculties
(
  FacultyID INT IDENTITY(1,1) PRIMARY KEY,
  FacultyName NVARCHAR(100) NOT NULL,
  DeanName NVARCHAR(100) NULL,
  Building NVARCHAR(100) NULL,
  UniversityID INT NOT NULL,
  CONSTRAINT FK_Faculties_Universities
    FOREIGN KEY (UniversityID)
    REFERENCES Universities(UniversityID)
);
GO

ابتدا Universities را بسازید و بعد Faculties را؛ چون جدول دوم به اولی وابسته است. در درج اطلاعات هم ترتیب همین است.

تمرین ۴: تعداد دانشکده‌های هر دانشگاه

SELECT
  U.UniversityName,
  COUNT(F.FacultyID) AS FacultyCount
FROM Universities AS U
LEFT JOIN Faculties AS F
  ON U.UniversityID = F.UniversityID
GROUP BY U.UniversityID, U.UniversityName;

COUNT می‌شمارد، GROUP BY دسته‌بندی می‌کند و LEFT JOIN دانشگاه‌های بدون دانشکده را با تعداد صفر نشان می‌دهد.

تمرین ۵: دانشگاه‌هایی با بیش از یک دانشکده

SELECT
  U.UniversityName,
  COUNT(F.FacultyID) AS FacultyCount
FROM Universities AS U
INNER JOIN Faculties AS F
  ON U.UniversityID = F.UniversityID
GROUP BY U.UniversityID, U.UniversityName
HAVING COUNT(F.FacultyID) > 1;

WHERE رکوردها را پیش از گروه‌بندی فیلتر می‌کند، اما HAVING روی گروه‌های ساخته‌شده شرط می‌گذارد.

تمرین ۱۱: بررسی صحت ارتباط دانشگاه و دانشکده

  F.FacultyID, F.FacultyName,
  F.UniversityID, U.UniversityName
FROM Faculties AS F
INNER JOIN Universities AS U
  ON F.UniversityID = U.UniversityID
ORDER BY U.UniversityName, F.FacultyName;

آزمایش رابطه یک‌به‌چند

SELECT U.UniversityName, F.FacultyName
FROM Universities AS U
LEFT JOIN Faculties AS F
  ON U.UniversityID = F.UniversityID
ORDER BY U.UniversityName, F.FacultyName;

نکات مهم اجرا

  • در UPDATE و DELETE حتماً از WHERE استفاده کنید؛ وگرنه همه رکوردها تغییر می‌کنند یا حذف می‌شوند.
  • دستورهای ساخت پایگاه داده و جدول را دوباره اجرا نکنید (خطای تکراری بودن).
  • با اجرای مجدد INSERT، شناسه‌های خودکار از ۱ شروع نمی‌شوند؛ قبل از استفاده از شناسه ثابت جدول را بررسی کنید.
  • حذف دانشگاهی که هنوز دانشکده دارد، به‌خاطر کلید خارجی اجرا نمی‌شود.
کار عملی ۴

مدیریت گروه‌های آموزشی

هر دانشکده می‌تواند چند گروه آموزشی داشته باشد (مثلاً مهندسی کامپیوتر، برق و عمران در دانشکده مهندسی). این کار روی پایگاه داده UniversityFacultiesDB از کار عملی ۳ ساخته می‌شود و نیازی به ساخت دوباره Faculties نیست.

Faculties (1) ───────< Departments (N)

ایجاد جدول Departments

USE UniversityFacultiesDB;
GO
CREATE TABLE Departments
(
  DepartmentID INT IDENTITY(1,1) PRIMARY KEY,
  DepartmentName NVARCHAR(100) NOT NULL,
  HeadName NVARCHAR(100) NULL,
  OfficeLocation NVARCHAR(100) NULL,
  FacultyID INT NOT NULL,
  CONSTRAINT FK_Departments_Faculties
    FOREIGN KEY (FacultyID)
    REFERENCES Faculties(FacultyID)
);
GO

REFERENCES Faculties(FacultyID) مشخص می‌کند شناسه دانشکده باید به رکوردی موجود اشاره کند.

کار عملی ۵

مدیریت رشته‌های تحصیلی و دانشجویان

ادامه پایگاه داده دانشگاه با دو جدول جدید Majors و Students. جدول‌های Universities، Faculties و Departments باید از قبل در UniversityFacultiesDB وجود داشته باشند.

Universities → Faculties → Departments → Majors → Students

هر رشته می‌تواند چند دانشجو داشته باشد و هر دانشجو به یک رشته وابسته است (یک‌به‌چند). آدرس دانشجو در صورت نیاز یک صفت در جدول Students است و موجودیت جدا نمی‌خواهد.

ایجاد جدول Majors

USE UniversityFacultiesDB;
GO
CREATE TABLE Majors
(
  MajorID INT IDENTITY(1,1) PRIMARY KEY,
  MajorName NVARCHAR(100) NOT NULL,
  DegreeLevel NVARCHAR(50) NOT NULL,
  DepartmentID INT NOT NULL,
  CONSTRAINT FK_Majors_Departments
    FOREIGN KEY (DepartmentID)
    REFERENCES Departments(DepartmentID)
);
GO
  • DegreeLevel: مقطع تحصیلی (کاردانی، کارشناسی، کارشناسی ارشد).
  • DepartmentID: کلید خارجی؛ هر رشته باید به یک گروه آموزشی موجود وابسته باشد.

ثبت رشته‌های نمونه

SELECT DepartmentID, DepartmentName FROM Departments;

INSERT INTO Majors (MajorName, DegreeLevel, DepartmentID)
VALUES
  (N'مهندسی نرم‌افزار', N'کارشناسی',      1),
  (N'فناوری اطلاعات',   N'کارشناسی',      1),
  (N'مهندسی کامپیوتر',  N'کارشناسی ارشد', 1),
  (N'مهندسی برق',       N'کارشناسی',      2),
  (N'مهندسی عمران',     N'کارشناسی',      3);
GO

شناسه‌های گروه را با داده‌های واقعی جدول Departments تطبیق دهید.

ایجاد جدول Students

CREATE TABLE Students
(
  StudentID INT IDENTITY(1,1) PRIMARY KEY,
  FirstName NVARCHAR(50) NOT NULL,
  LastName NVARCHAR(50) NOT NULL,
  NationalCode CHAR(10) NOT NULL UNIQUE,
  BirthDate DATE NULL,
  EntryYear INT NOT NULL,
  MajorID INT NOT NULL,
  CONSTRAINT FK_Students_Majors
    FOREIGN KEY (MajorID)
    REFERENCES Majors(MajorID)
);
GO

کد ملی یکتاست (UNIQUE) و EntryYear سال ورود به دانشگاه است.

کار عملی ۶

مدیریت درس‌ها و انتخاب واحد

ارتباط دانشجو و درس چندبه‌چند است: هر دانشجو چند درس می‌گیرد و هر درس را چند دانشجو انتخاب می‌کنند. برای پیاده‌سازی آن از جدول واسط StudentCourses استفاده می‌شود. هر درس هم به یک گروه آموزشی وابسته است.

Students (1) ──< StudentCourses >── (1) Courses

ایجاد جدول Courses

USE UniversityFacultiesDB;
GO
CREATE TABLE Courses
(
  CourseID INT IDENTITY(1,1) PRIMARY KEY,
  CourseCode NVARCHAR(20) NOT NULL UNIQUE,
  CourseName NVARCHAR(100) NOT NULL,
  Units INT NOT NULL,
  DepartmentID INT NOT NULL,
  CONSTRAINT CK_Courses_Units
    CHECK (Units BETWEEN 1 AND 6),
  CONSTRAINT FK_Courses_Departments
    FOREIGN KEY (DepartmentID)
    REFERENCES Departments(DepartmentID)
);
GO

CHECK تعداد واحد درس را بین ۱ تا ۶ محدود می‌کند و CourseCode باید یکتا باشد.

ایجاد جدول واسط StudentCourses

CREATE TABLE StudentCourses
(
  EnrollmentID INT IDENTITY(1,1) PRIMARY KEY,
  StudentID INT NOT NULL,
  CourseID INT NOT NULL,
  AcademicYear INT NOT NULL,
  Semester NVARCHAR(20) NOT NULL,
  Grade DECIMAL(4,2) NULL,
  CONSTRAINT FK_StudentCourses_Students
    FOREIGN KEY (StudentID) REFERENCES Students(StudentID),
  CONSTRAINT FK_StudentCourses_Courses
    FOREIGN KEY (CourseID) REFERENCES Courses(CourseID),
  CONSTRAINT CK_StudentCourses_Grade
    CHECK (Grade IS NULL OR Grade BETWEEN 0 AND 20),
  CONSTRAINT UQ_StudentCourses
    UNIQUE (StudentID, CourseID, AcademicYear, Semester)
);
GO
  • دو کلید خارجی (StudentID و CourseID) دانشجو و درس را به هم وصل می‌کنند.
  • Grade می‌تواند در ابتدا NULL باشد و فقط مقدار ۰ تا ۲۰ می‌پذیرد.
  • UNIQUE مانع ثبت تکراری یک درس برای یک دانشجو در یک نیم‌سال و سال تحصیلی می‌شود.
